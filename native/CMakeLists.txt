cmake_minimum_required(VERSION 3.28)

target_include_directories(serverdeck_native PUBLIC include)
target_compile_definitions(serverdeck_native PUBLIC UNICODE _UNICODE NOMINMAX)
target_compile_options(serverdeck_native PRIVATE /W4 /WX /EHsc)
target_link_libraries(serverdeck_native PUBLIC bcrypt advapi32 ole32 shell32 user32 crypt32 winhttp ws2_32 uuid shlwapi)

  endif()
  configure_file(bootstrap/build-version.txt.in "${CMAKE_CURRENT_BINARY_DIR}/build-version.txt" @ONLY NEWLINE_STYLE UNIX)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/build-version.rc" "SERVERDECK_BUILD_V1 RCDATA \"${CMAKE_CURRENT_BINARY_DIR}/build-version.txt\"\n")
  add_executable(ServerDeck WIN32 bootstrap/main.cpp "${CMAKE_CURRENT_BINARY_DIR}/app_icon.rc" "${CMAKE_CURRENT_BINARY_DIR}/build-version.rc" ${SERVERDECK_PAYLOAD_RESOURCE})
  target_include_directories(ServerDeck PRIVATE include)
  target_compile_definitions(ServerDeck PRIVATE UNICODE _UNICODE NOMINMAX)
  target_compile_options(ServerDeck PRIVATE /W4 /WX /EHsc)
  target_link_libraries(ServerDeck PRIVATE serverdeck_native)
  if(SERVERDECK_TRUST_RESOURCE)
    target_sources(ServerDeck PRIVATE ${SERVERDECK_TRUST_RESOURCE})
  endif()
endif()
