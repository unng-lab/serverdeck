cmake_minimum_required(VERSION 3.28)
project(serverdeck_bootstrap LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "ServerDeck requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/utf-8>")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(SERVERDECK_BUILD_TESTS "Build native unit tests" OFF)
option(SERVERDECK_BUILD_HISTORICAL_INTEGRATION "Build retired integration helpers (not unit tests)" OFF)
set(SERVERDECK_PAYLOAD_RESOURCE "" CACHE FILEPATH "Generated payload RC file")
configure_file(bootstrap/app_icon.rc.in "${CMAKE_CURRENT_BINARY_DIR}/app_icon.rc" @ONLY)
add_library(serverdeck_native STATIC bootstrap/security.cpp bootstrap/payload.cpp bootstrap/instance.cpp bootstrap/mcp_pipe.cpp)
target_include_directories(serverdeck_native PUBLIC include)
target_compile_definitions(serverdeck_native PUBLIC UNICODE _UNICODE NOMINMAX)
target_compile_options(serverdeck_native PRIVATE /W4 /WX /EHsc)
target_link_libraries(serverdeck_native PUBLIC bcrypt advapi32 ole32 shell32 user32 crypt32)
if(SERVERDECK_PAYLOAD_RESOURCE)
  add_executable(ServerDeck WIN32 bootstrap/main.cpp "${CMAKE_CURRENT_BINARY_DIR}/app_icon.rc" ${SERVERDECK_PAYLOAD_RESOURCE})
  target_include_directories(ServerDeck PRIVATE include)
  target_compile_definitions(ServerDeck PRIVATE UNICODE _UNICODE NOMINMAX)
  target_compile_options(ServerDeck PRIVATE /W4 /WX /EHsc)
  target_link_libraries(ServerDeck PRIVATE serverdeck_native)
endif()
if(SERVERDECK_BUILD_TESTS)
  enable_testing()
  add_executable(app_icon_test tests/app_icon_test.cpp "${CMAKE_CURRENT_BINARY_DIR}/app_icon.rc")
  target_compile_definitions(app_icon_test PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(app_icon_test PRIVATE user32)
  add_test(NAME app_icon COMMAND app_icon_test)
  set_tests_properties(app_icon PROPERTIES LABELS "unit")
  add_executable(runtime_contract_test tests/runtime_contract_test.cpp)
  target_include_directories(runtime_contract_test PRIVATE include)
  add_test(NAME runtime_contract COMMAND runtime_contract_test)
  add_executable(ipc_auth_test tests/ipc_auth_test.cpp)
  target_include_directories(ipc_auth_test PRIVATE include)
  target_compile_options(ipc_auth_test PRIVATE /W4 /WX /EHsc)
  add_test(NAME ipc_auth COMMAND ipc_auth_test)
  set_tests_properties(ipc_auth PROPERTIES LABELS "unit")
  add_executable(ipc_boundary_test tests/ipc_boundary_test.cpp)
  target_link_libraries(ipc_boundary_test PRIVATE serverdeck_native)
  target_compile_options(ipc_boundary_test PRIVATE /W4 /WX /EHsc)
  add_test(NAME ipc_boundary COMMAND ipc_boundary_test)
  set_tests_properties(ipc_boundary PROPERTIES LABELS "unit" TIMEOUT 15)
  add_executable(payload_test tests/payload_test.cpp)
  target_link_libraries(payload_test PRIVATE serverdeck_native)
  add_test(NAME payload COMMAND payload_test)
  add_executable(instance_test tests/instance_test.cpp)
  target_link_libraries(instance_test PRIVATE serverdeck_native)
  add_test(NAME instance COMMAND instance_test)
  set_tests_properties(instance PROPERTIES TIMEOUT 15)
  add_executable(security_test tests/security_test.cpp)
  target_link_libraries(security_test PRIVATE serverdeck_native)
  add_test(NAME security COMMAND security_test)
  add_executable(runtime_owners_test tests/runtime_owners_test.cpp)
  target_include_directories(runtime_owners_test PRIVATE include)
  add_test(NAME runtime_owners COMMAND runtime_owners_test)
  set(SERVERDECK_WRAPPER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../windows/flutter/ephemeral/cpp_client_wrapper"
      CACHE PATH "Generated pinned Flutter C++ codec sources")
  if(NOT EXISTS "${SERVERDECK_WRAPPER_ROOT}/standard_codec.cc")
    message(FATAL_ERROR "Run flutter build windows first to prepare the pinned C++ codec sources")
  endif()
  # The synthetic messenger needs only the codec, not Flutter engine imports.
  # /MT keeps this verification helper runnable on a clean no-VC VM.
  add_executable(platform_channel_test tests/platform_channel_test.cpp
    ../windows/runner/platform_channel.cpp "${SERVERDECK_WRAPPER_ROOT}/standard_codec.cc"
    "${SERVERDECK_WRAPPER_ROOT}/core_implementations.cc")
  target_include_directories(platform_channel_test PRIVATE
    "${SERVERDECK_WRAPPER_ROOT}/include" "${SERVERDECK_WRAPPER_ROOT}/..")
  target_compile_options(platform_channel_test PRIVATE /W4 /WX /wd4100 /EHsc /Gy /Gw)
  target_link_options(platform_channel_test PRIVATE /OPT:REF)
  target_link_libraries(platform_channel_test PRIVATE serverdeck_native crypt32
    "${SERVERDECK_WRAPPER_ROOT}/../flutter_windows.dll.lib")
  add_test(NAME platform_channel COMMAND platform_channel_test)
  set_tests_properties(platform_channel PROPERTIES TIMEOUT 15)
  set_tests_properties(runtime_contract payload instance security runtime_owners platform_channel
    PROPERTIES LABELS "unit")
  if(SERVERDECK_BUILD_HISTORICAL_INTEGRATION)
  add_executable(release_observer tests/release_observer.cpp)
  target_compile_options(release_observer PRIVATE /W4 /WX /EHsc)
  target_link_libraries(release_observer PRIVATE serverdeck_native psapi)
  add_test(NAME release_observer COMMAND release_observer --self-test)
  set_tests_properties(release_observer PROPERTIES TIMEOUT 15)
  add_executable(release_lifecycle tests/release_lifecycle.cpp)
  target_compile_options(release_lifecycle PRIVATE /W4 /WX /EHsc)
  target_link_libraries(release_lifecycle PRIVATE serverdeck_native psapi oleaut32 wbemuuid)
  add_test(NAME release_lifecycle COMMAND release_lifecycle --self-test)
  set_tests_properties(release_lifecycle PROPERTIES TIMEOUT 15)
  set_tests_properties(release_observer release_lifecycle PROPERTIES LABELS "historical-integration")
  endif()
endif()
